\section{Experimental Plan and Architecture}
\label{sec:experimental_plan}

\paragraph{Experimental Setting}
All experiments execute without neural network gradient descent, backpropagation, or parameter optimization. The post-processing pipeline operates on frozen representations extracted in a single forward pass of the models described in Section~\ref{sec:baselines_sota}. Random seeds across \texttt{torch}, \texttt{numpy}, and Python \texttt{random} are fixed to $2026$. Hardware and numerical precision are listed in Table~\ref{tab:compute_resources}.

\paragraph{Post-Processing Transformations}
Let $x \in \mathbb{R}^d$ be an embedding produced by one model, where $d$ is the model's native dimension. Every transformation is fitted on a calibration set $X = \{x_1, \dots, x_N\}$: the document embeddings for retrieval, and the embeddings of both sentence sets for STS (Section~\ref{sec:datasets}). From $X$ we estimate the mean vector
\begin{equation}
\mu = \frac{1}{N} \sum_{i=1}^{N} x_i,
\end{equation}
the standard deviation $\sigma_j$ of each coordinate $j$,
\begin{equation}
\sigma_j = \sqrt{\frac{1}{N-1} \sum_{i=1}^{N} (x_{ij} - \mu_j)^2},
\end{equation}
and the covariance matrix with its eigendecomposition
\begin{equation}
\Sigma = \frac{1}{N-1} \sum_{i=1}^{N} (x_i - \mu)(x_i - \mu)^\top = U \Lambda U^\top .
\end{equation}
Here $\Lambda = \mathrm{diag}(\lambda_1, \dots, \lambda_d)$ holds the eigenvalues in descending order, and the columns $u_1, \dots, u_d$ of $U$ are the matching eigenvectors. $U_k$ denotes the first $k$ columns of $U$, and $\Lambda_k$ the top-left $k \times k$ block of $\Lambda$. Unless stated otherwise, every output is projected onto the unit sphere by
\begin{equation}
\phi(z) = z / \|z\|_2 .
\end{equation}

The full-dimension transformations ($d \to d$) are as follows. The \texttt{baseline} only normalizes the raw embedding:
\begin{equation}
f_{\text{base}}(x) = \phi(x).
\end{equation}
\texttt{standardization} \citep{timkey2021all} rescales each coordinate to zero mean and unit variance, where $\oslash$ denotes element-wise division:
\begin{equation}
f_{\text{std}}(x) = \phi\big((x - \mu) \oslash \sigma\big).
\end{equation}
\texttt{r1} \citep{ren2025correcting} subtracts the mean:
\begin{equation}
f_{\text{R1}}(x) = \phi(x - \mu).
\end{equation}
\texttt{r2} \citep{ren2025correcting} instead removes the component of $x$ along the unit mean direction $\hat{\mu} = \mu / \|\mu\|_2$:
\begin{equation}
f_{\text{R2}}(x) = \phi\big(x - (x^\top \hat{\mu})\, \hat{\mu}\big).
\end{equation}
\texttt{abtt\_D} \citep{mu2018all} centers $x$ and removes its projection onto the top $D \in \{1, 2, 3\}$ principal directions:
\begin{equation}
f_{\text{ABTT-}D}(x) = \phi\big((I - U_D U_D^\top)(x - \mu)\big).
\end{equation}
\texttt{soft\_zca} \citep{diera2024isotropy} whitens with a regularized inverse square root of the covariance, using $\epsilon = 10^{-4}$:
\begin{equation}
W_{\text{ZCA}} = U (\Lambda + \epsilon I)^{-1/2} U^\top,
\end{equation}
\begin{equation}
f_{\text{ZCA}}(x) = \phi\big(W_{\text{ZCA}} (x - \mu)\big).
\end{equation}

Two negative controls follow \citet{ren2025correcting}. \texttt{rand} repeats R2 with a random unit direction $r = g / \|g\|_2$, where $g \sim \mathcal{N}(0, I_d)$ is drawn with a fixed seed:
\begin{equation}
f_{\text{rand}}(x) = \phi\big(x - (x^\top r)\, r\big).
\end{equation}
\texttt{mc} repeats R1 without the final normalization:
\begin{equation}
f_{\text{mc}}(x) = x - \mu.
\end{equation}

The compression transformations map $d \to k$ with $k \in \{512, 256, 128, 64\}$. \texttt{prefix} keeps the first $k$ coordinates, as in Matryoshka Representation Learning \citep{kusupati2022matryoshka}:
\begin{equation}
f_{\text{prefix}}(x) = \phi(x_{1:k}).
\end{equation}
\texttt{random\_truncation} (control) keeps a fixed random subset $S \subset \{1, \dots, d\}$ of $|S| = k$ coordinates, drawn uniformly without replacement:
\begin{equation}
f_{\text{trunc}}(x) = \phi(x_S).
\end{equation}
\texttt{pca}, \texttt{whitening}, and \texttt{spectemp} share one spectral form \citep{li2026spectral} with exponent $\gamma \in [0, 1]$ and $\epsilon' = 10^{-6}$:
\begin{equation}
f_{\gamma}(x) = \phi\big((\Lambda_k + \epsilon' I)^{-\gamma/2}\, U_k^\top (x - \mu)\big).
\end{equation}
\texttt{pca} uses $\gamma = 0$, an unweighted projection onto the top $k$ principal directions. \texttt{whitening} uses $\gamma = 1$, which equalizes the variance of all $k$ retained directions \citep{su2021whitening}. \texttt{spectemp} chooses $\gamma$ for each $k$ from the eigenvalue spectrum \citep{li2026spectral}. It takes the noise floor $\eta$ as the mean of the smallest 10\% of eigenvalues and defines the signal-to-noise ratio of each remaining eigenvalue as
\begin{equation}
\mathrm{SNR}(i) = \max\!\left(0, \frac{\lambda_i - \eta}{\eta}\right).
\end{equation}
The knee $k_{\text{knee}}$ of this curve is found with the Kneedle algorithm \citep{satopaa2011finding} with sensitivity $S = 0.5$, and
\begin{equation}
\gamma(k) = \min\!\left(1, \frac{\mathrm{SNR}(k)}{\mathrm{SNR}(k_{\text{knee}})}\right).
\end{equation}

\paragraph{Evaluation Metrics and Rationale}
\begin{itemize}
\item Retrieval: Evaluated using \texttt{pytrec-eval} against official qrels. Primary metric is \texttt{ndcg\_at\_10} (evaluating graded relevance order). Secondary metrics are \texttt{recall\_at\_100} (candidate coverage) and \texttt{mrr\_at\_10} (reciprocal rank of first relevant passage).
\item Similarity: Primary metric is Spearman rank correlation $\rho \times 100$ (measuring monotonic ranking agreement). Secondary metric is Pearson linear correlation $r \times 100$.
\item Intrinsic Geometry: Measured on the transformed embeddings, with $\mu$ and $\lambda_j$ recomputed from them: centroid norm ($\|\mu\|_2$), average pairwise cosine similarity across 2,000 sampled embeddings, Maximum Explainable Variance of the leading eigenvalue ($\text{MEV} = \lambda_1 / \sum_j \lambda_j$), Normalized Intrinsic Dimensionality ($\text{NID} = H / \log d$, where $H = -\sum_j p_j \log p_j$ and $p_j = \lambda_j / \sum_l \lambda_l$) \citep{yokoi2024word}, and IsoScore \citep{rudman-etal-2022-isoscore}.
\end{itemize}
